cmake_minimum_required(VERSION 3.16)
project(ultra_native_x86_module LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
add_library(native_x86_module MODULE src/module/native_x86_module.cpp)
target_include_directories(native_x86_module PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../include)
if(NOT WIN32)
    target_link_libraries(native_x86_module PRIVATE dl)
endif()
add_executable(test_host tests/test_host.cpp src/module/native_x86_module.cpp)
target_include_directories(test_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../include)
if(NOT WIN32)
    target_link_libraries(test_host PRIVATE dl)
endif()
enable_testing()
add_test(NAME abi_lifecycle COMMAND test_host)
